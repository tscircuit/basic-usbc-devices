cmake_minimum_required(VERSION 3.16)
set(PICO_BOARD solderparty_rp2350_stamp CACHE STRING "Embedded host module")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(motherboard C CXX ASM)
pico_sdk_init()
add_executable(motherboard main.c)
target_include_directories(motherboard PRIVATE .)
target_link_libraries(motherboard pico_stdlib tinyusb_host hardware_uart)
pico_enable_stdio_usb(motherboard 0)
pico_enable_stdio_uart(motherboard 1)
pico_add_extra_outputs(motherboard)
